```js

//代码如果放在main.js中
//开启无障碍服务
 startEnv()
//脚本停止取消监听
setStopCallback(function () {
        cancelObserveEvent("key-up")
})
//监听音量下键
observeEvent("key-up", function (key, data) {
    if (JSON.parse(data).keyCode === 25) {
    	logd("脚本停止")
        exit()
    }
})
//模拟脚本运行
while (true) {
    sleep(1000);
}


//代码如果放在ui.js中

//只监听音量下键就行了
observeEvent("key-up", function (key, data) {
    if (JSON.parse(data).keyCode === 25) {
        logd("脚本停止")
        exit()
    }
})
```
